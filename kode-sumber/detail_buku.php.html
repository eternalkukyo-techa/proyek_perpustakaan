<?php
session_start();
include 'koneksi.php';

$id = isset($_GET['id']) ? (int)$_GET['id'] : 0;
$buku = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT * FROM bibliografi WHERE id_buku=$id"));

if (!$buku) { echo "<script>alert('Buku tidak ditemukan!'); window.location='opac.php';</script>"; exit; }

// PROSES AJUKAN PINJAM
$pesan = '';
if (isset($_POST['ajukan_pinjam'])) {
    if (!isset($_SESSION['login']) || $_SESSION['role'] != 'user') {
        $pesan = "Anda harus login sebagai Pengguna untuk meminjam buku!";
    } else {
        $username = $_SESSION['username'];
        $anggota = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT id_anggota FROM anggota WHERE username='$username'"));
        
        if ($anggota) {
            $id_anggota = $anggota['id_anggota'];
            $tgl_pinjam = date('Y-m-d');
            $tgl_kembali = date('Y-m-d', strtotime('+7 days'));

            // Cek apakah user ini sudah mengajukan/meminjam buku yang sama
            $cek = mysqli_query($koneksi, "SELECT * FROM sirkulasi WHERE id_anggota=$id_anggota AND id_buku=$id AND status IN('Menunggu','Dipinjam')");
            if (mysqli_num_rows($cek) > 0) {
                $pesan = "Anda sudah meminjam atau sedang mengajukan buku ini.";
            } else {
                // Insert ke database dengan status MENUNGGU (stok belum berkurang)
                mysqli_query($koneksi, "INSERT INTO sirkulasi (id_anggota, id_buku, tanggal_pinjam, tanggal_kembali_seharusnya, status) VALUES ($id_anggota, $id, '$tgl_pinjam', '$tgl_kembali', 'Menunggu')");
                $pesan = "Berhasil! Pengajuan pinjaman Anda sedang menunggu ACC dari Pustakawan.";
            }
        }
    }
}

// CEK STATUS PEMINJAMAN USER SAAT INI
$status_user = '';
if (isset($_SESSION['login']) && $_SESSION['role'] == 'user') {
    $username = $_SESSION['username'];
    $anggota = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT id_anggota FROM anggota WHERE username='$username'"));
    if ($anggota) {
        $id_anggota = $anggota['id_anggota'];
        $cek_trx = mysqli_query($koneksi, "SELECT status FROM sirkulasi WHERE id_anggota=$id_anggota AND id_buku=$id AND status IN('Menunggu','Dipinjam') ORDER BY id_sirkulasi DESC LIMIT 1");
        if (mysqli_num_rows($cek_trx) > 0) {
            $status_user = mysqli_fetch_assoc($cek_trx)['status'];
        }
    }
}
