    $id = (int)$_GET['acc'];
    $trx = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT * FROM sirkulasi WHERE id_sirkulasi=$id AND status='Menunggu'"));
    if ($trx) {
        $id_buku = $trx['id_buku'];
        $stok = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT stok FROM bibliografi WHERE id_buku=$id_buku"))['stok'];
        if ($stok > 0) {
            mysqli_query($koneksi, "UPDATE sirkulasi SET status='Dipinjam', alasan_tolak=NULL WHERE id_sirkulasi=$id");
            mysqli_query($koneksi, "UPDATE bibliografi SET stok=stok-1 WHERE id_buku=$id_buku");
        }
        header("Location: dashboard.php"); exit;
    }
}

        function bukaModalTolakDash(btn) {
