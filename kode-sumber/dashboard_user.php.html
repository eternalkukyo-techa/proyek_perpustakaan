    $tgl_kembali = date('Y-m-d', strtotime('+7 days'));

    $cek = mysqli_query($koneksi, "SELECT * FROM sirkulasi WHERE id_anggota=$id_anggota AND id_buku=$id_buku AND status IN('Menunggu','Dipinjam')");
    if (mysqli_num_rows($cek) > 0) {
        $pesan = "Anda sudah meminjam atau sedang mengajukan buku ini.";
    } else {
        mysqli_query($koneksi, "INSERT INTO sirkulasi (id_anggota, id_buku, tanggal_pinjam, tanggal_kembali_seharusnya, status) VALUES ($id_anggota, $id_buku, '$tgl_pinjam', '$tgl_kembali', 'Menunggu')");

    if ($trx) {
        $tgl_skrg = date('Y-m-d');
        $jatuh_tempo = strtotime($trx['tanggal_kembali_seharusnya']);
        $tgl_dikembalikan = strtotime($tgl_skrg);
        
        // Hitung denda otomatis jika telat (Rp 1.000 / hari)
        $denda = 0;
        if ($tgl_dikembalikan > $jatuh_tempo) {
            $selisih_hari = floor(($tgl_dikembalikan - $jatuh_tempo) / (60 * 60 * 24));
            $denda = $selisih_hari * 1000;
        }
        
        // Update transaksi jadi Kembali
        mysqli_query($koneksi, "UPDATE sirkulasi SET tanggal_dikembalikan='$tgl_skrg', status='Kembali', denda=$denda WHERE id_sirkulasi=$id_sirkulasi");
        
        // Kembalikan Stok Buku
        mysqli_query($koneksi, "UPDATE bibliografi SET stok=stok+1 WHERE id_buku={$trx['id_buku']}");
        
        $pesan = "Buku berhasil dikembalikan!" . ($denda > 0 ? " Anda dikenakan denda Rp " . number_format($denda, 0, ',', '.') : " Terima kasih telah tepat waktu.");
        
        // Refresh halaman agar status berubah
        echo "<script>alert('$pesan'); window.location='dashboard_user.php';</script>";
        exit;
    }
