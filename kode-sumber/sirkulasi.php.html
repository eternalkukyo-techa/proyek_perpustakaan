<?php
session_start();
if (!isset($_SESSION['login']) || $_SESSION['role'] != 'admin') { header("Location: index.php"); exit; }
include 'koneksi.php';

$pesan = '';

// 1. PROSES ADMIN PINJAMKAN LANGSUNG (TANPA ACC)
if (isset($_POST['pinjam'])) {
    $id_anggota = (int)$_POST['id_anggota'];
    $id_buku = (int)$_POST['id_buku'];
    $tgl_pinjam = $_POST['tanggal_pinjam'];
    $lama = (int)$_POST['lama_pinjam'];
    $tgl_kembali_seharusnya = date('Y-m-d', strtotime($tgl_pinjam . " + $lama days"));

    $stok = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT stok FROM bibliografi WHERE id_buku=$id_buku"))['stok'];
    if ($stok > 0) {
        mysqli_query($koneksi, "INSERT INTO sirkulasi (id_anggota, id_buku, tanggal_pinjam, tanggal_kembali_seharusnya, status) VALUES ($id_anggota, $id_buku, '$tgl_pinjam', '$tgl_kembali_seharusnya', 'Dipinjam')");
        mysqli_query($koneksi, "UPDATE bibliografi SET stok=stok-1 WHERE id_buku=$id_buku");
        header("Location: sirkulasi.php"); exit;
    } else { $pesan = "Stok buku habis!"; }
}

// 2. PROSES ACC PENGAJUAN USER
if (isset($_GET['acc'])) {
    $id = (int)$_GET['acc'];
    $trx = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT * FROM sirkulasi WHERE id_sirkulasi=$id AND status='Menunggu'"));
    if ($trx) {
        $id_buku = $trx['id_buku'];
        $stok = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT stok FROM bibliografi WHERE id_buku=$id_buku"))['stok'];
        
        if ($stok > 0) {
            mysqli_query($koneksi, "UPDATE sirkulasi SET status='Dipinjam', alasan_tolak=NULL WHERE id_sirkulasi=$id");
            mysqli_query($koneksi, "UPDATE bibliografi SET stok=stok-1 WHERE id_buku=$id_buku");
            header("Location: sirkulasi.php"); exit;
        } else {
            $pesan = "Gagal ACC! Stok buku ini keburu habis.";
        }
    }
}

// 3. PROSES TOLAK PENGAJUAN (SIMPAN STATUS DITOLAK & ALASAN)
if (isset($_POST['tolak_pengajuan'])) {
    $id = (int)$_POST['id_sirkulasi'];
    $alasan = trim(mysqli_real_escape_string($koneksi, $_POST['alasan_tolak']));
    if ($alasan === '') {
        $alasan = 'Pengajuan peminjaman tidak dapat disetujui oleh pustakawan.';
    }

    mysqli_query($koneksi, "UPDATE sirkulasi SET status='Ditolak', alasan_tolak='$alasan' WHERE id_sirkulasi=$id AND status='Menunggu'");
    header("Location: sirkulasi.php"); exit;
}

// 4. PROSES KEMBALIKAN (TEPAT WAKTU / TERLAMBAT)
if (isset($_GET['kembali'])) {
    $id = (int)$_GET['kembali'];
    $denda = isset($_GET['denda']) ? (int)$_GET['denda'] : 0;
    $tgl_skrg = date('Y-m-d');
    
    $trx = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT * FROM sirkulasi WHERE id_sirkulasi=$id"));
    if ($trx && $trx['status'] == 'Dipinjam') {
        mysqli_query($koneksi, "UPDATE sirkulasi SET tanggal_dikembalikan='$tgl_skrg', status='Kembali', denda=$denda WHERE id_sirkulasi=$id");
        mysqli_query($koneksi, "UPDATE bibliografi SET stok=stok+1 WHERE id_buku={$trx['id_buku']}");
        header("Location: sirkulasi.php"); exit;
    }
}

    <script>
        function bukaModalTolak(btn) {
            document.getElementById('tolak_id').value = btn.getAttribute('data-id');
            document.getElementById('tolak_judul').textContent = btn.getAttribute('data-judul');
            document.getElementById('tolak_nama').textContent = btn.getAttribute('data-nama');
            document.getElementById('alasan_tolak').value = '';
            document.getElementById('modalTolak').classList.add('show');
            document.getElementById('alasan_tolak').focus();
        }

        function tutupModalTolak() {
            document.getElementById('modalTolak').classList.remove('show');
        }

        function tutupModalBackdrop(e) {
            if (e.target.id === 'modalTolak') {
                tutupModalTolak();
            }
        }

        function kembalikanTepat(id) {
            if(confirm('Proses pengembalian buku tepat waktu? (Denda Rp 0)')) {
                window.location.href = 'sirkulasi.php?kembali=' + id + '&denda=0';
            }
        }
        function kembalikanTelat(id) {
            let nominal = prompt("Buku terlambat! Masukkan nominal denda (hanya angka, contoh: 5000):", "5000");
            if (nominal != null && nominal != "" && !isNaN(nominal)) {
                window.location.href = 'sirkulasi.php?kembali=' + id + '&denda=' + nominal;
            }
        }
