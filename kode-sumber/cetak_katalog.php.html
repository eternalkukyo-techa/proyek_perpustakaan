// Helper untuk membuat kode cutter / nomor panggil otomatis jika kolom khusus tidak ada
function buatNomorPanggil(array $row): array {
    if (!empty($row['no_panggil'])) {
        $baris = preg_split('/\r\n|\r|\n/', trim($row['no_panggil']));
        return [
            'kelas'  => $baris[0] ?? '',
            'cutter' => $baris[1] ?? '',
            'awal'   => $baris[2] ?? ''
        ];
    }

    $pengarang = trim($row['pengarang'] ?? '');
    $judul = trim($row['judul'] ?? '');
    $kategori = trim($row['kategori'] ?? '');

    $kelasMap = [
        'Fiksi'     => '800',
        'Fantasi'   => '808',
        'Anak-Anak' => '028.5',
        'Referensi' => '011'
    ];
    $kelas = $kelasMap[$kategori] ?? '020';

    $sumberHuruf = $pengarang !== '' ? $pengarang : $judul;
    $cutter = strtoupper(substr(preg_replace('/[^A-Za-z]/', '', $sumberHuruf), 0, 3));
    if ($cutter === '') {
        $cutter = 'KOR';
    }

    $awalHuruf = strtolower(substr(preg_replace('/[^A-Za-z]/', '', $judul), 0, 1));
    if ($awalHuruf === '') {
        $awalHuruf = 'k';
    }

    return [
        'kelas'  => $kelas,
        'cutter' => $cutter,
        'awal'   => $awalHuruf
    ];
}
