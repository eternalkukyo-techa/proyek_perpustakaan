    <script>
        function filterBooks() {
            let input = document.getElementById("liveSearch").value.toLowerCase();
            let filter = document.getElementById("filterKriteria").value;
            let cards = document.getElementsByClassName("book-card");

            for (let i = 0; i < cards.length; i++) {
                let card = cards[i];
                let judul = card.getAttribute("data-judul");
                let pengarang = card.getAttribute("data-pengarang");
                let penerbit = card.getAttribute("data-penerbit");
                
                let match = false;

                if (filter === 'all') {
                    match = judul.includes(input) || pengarang.includes(input) || penerbit.includes(input);
                } else if (filter === 'judul') {
                    match = judul.includes(input);
                } else if (filter === 'pengarang') {
                    match = pengarang.includes(input);
                } else if (filter === 'penerbit') {
                    match = penerbit.includes(input);
                }

                if (match) {
                    card.style.display = "flex";
                } else {
                    card.style.display = "none";
                }
            }
        }
    </script>
