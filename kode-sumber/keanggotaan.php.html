if (!isset($_SESSION['login']) || $_SESSION['role'] != 'admin') { header("Location: index.php"); exit; }

if (isset($_POST['acc_anggota'])) {
    $id = (int)$_POST['id_anggota'];
    $nomor = mysqli_real_escape_string($koneksi, $_POST['nomor_anggota']);
    
    // Validasi apakah nomor anggota sudah dipakai
    $cek = mysqli_query($koneksi, "SELECT * FROM anggota WHERE nomor_anggota='$nomor'");
    if (mysqli_num_rows($cek) > 0) {
        echo "<script>alert('Gagal! Nomor Anggota $nomor sudah dipakai anggota lain.');</script>";
    } else {
        mysqli_query($koneksi, "UPDATE anggota SET status='Aktif', nomor_anggota='$nomor' WHERE id_anggota=$id");
        header("Location: keanggotaan.php"); exit;
    }
}

// PROSES TOLAK (HAPUS) PENDAFTAR
if (isset($_GET['tolak'])) {
    $id = (int)$_GET['tolak'];
    mysqli_query($koneksi, "DELETE FROM anggota WHERE id_anggota=$id");
    header("Location: keanggotaan.php"); exit;
}

// PROSES TAMBAH MANUAL (OLEH ADMIN)

    $nomor = mysqli_real_escape_string($koneksi, $_POST['nomor_anggota']);
    $nama = mysqli_real_escape_string($koneksi, $_POST['nama']);
    $jk = mysqli_real_escape_string($koneksi, $_POST['jenis_kelamin']);
    $alamat = mysqli_real_escape_string($koneksi, $_POST['alamat']);
    $hp = mysqli_real_escape_string($koneksi, $_POST['no_hp']);
    $tgl = date('Y-m-d');
    $auto_user = strtolower(str_replace(' ', '', $nomor)); 
    $auto_pass = '123456'; 

    mysqli_query($koneksi, "INSERT INTO anggota (nomor_anggota, nama, jenis_kelamin, alamat, no_hp, tanggal_daftar, username, password, status) VALUES ('$nomor', '$nama', '$jk', '$alamat', '$hp', '$tgl', '$auto_user', '$auto_pass', 'Aktif')");
    header("Location: keanggotaan.php"); exit;
}

// PROSES EDIT ANGGOTA (TERMASUK USERNAME & PASSWORD)

    $id = (int)$_POST['id_anggota'];
    $nomor = mysqli_real_escape_string($koneksi, $_POST['nomor_anggota']);
    $nama = mysqli_real_escape_string($koneksi, $_POST['nama']);
    $jk = mysqli_real_escape_string($koneksi, $_POST['jenis_kelamin']);
    $alamat = mysqli_real_escape_string($koneksi, $_POST['alamat']);
    $hp = mysqli_real_escape_string($koneksi, $_POST['no_hp']);
    $username = mysqli_real_escape_string($koneksi, $_POST['username']);
    $password = mysqli_real_escape_string($koneksi, $_POST['password']);

    // Cek apakah nomor anggota dipakai anggota lain
    $cek_nomor = mysqli_query($koneksi, "SELECT id_anggota FROM anggota WHERE nomor_anggota='$nomor' AND id_anggota != $id");
    // Cek apakah username dipakai anggota lain
    $cek_user = mysqli_query($koneksi, "SELECT id_anggota FROM anggota WHERE username='$username' AND id_anggota != $id");

    if (mysqli_num_rows($cek_nomor) > 0) {
        echo "<script>alert('Gagal! Nomor Anggota $nomor sudah digunakan anggota lain.');</script>";
    } elseif (mysqli_num_rows($cek_user) > 0) {
        echo "<script>alert('Gagal! Username $username sudah digunakan anggota lain.');</script>";
    } else {
        // Jika kolom password dikosongkan, pertahankan password lama
        if (!empty($password)) {
            $query_edit = "UPDATE anggota SET 
                nomor_anggota='$nomor', 
                nama='$nama', 
                jenis_kelamin='$jk', 
                alamat='$alamat', 
                no_hp='$hp', 
                username='$username', 
                password='$password' 
                WHERE id_anggota=$id";
        } else {
            $query_edit = "UPDATE anggota SET 
                nomor_anggota='$nomor', 
                nama='$nama', 
                jenis_kelamin='$jk', 
                alamat='$alamat', 
                no_hp='$hp', 
                username='$username' 
                WHERE id_anggota=$id";
        }

        if (mysqli_query($koneksi, $query_edit)) {
            header("Location: keanggotaan.php"); exit;
        } else {
            echo "<script>alert('Terjadi kesalahan saat mengupdate data.');</script>";
        }
    }
}
