<?php
session_start();
include 'koneksi.php';

$error = '';
$sukses = '';

// PROSES REGISTRASI 
if (isset($_POST['register'])) {
    $nama = mysqli_real_escape_string($koneksi, $_POST['nama']);
    $jk = mysqli_real_escape_string($koneksi, $_POST['jenis_kelamin']);
    $alamat = mysqli_real_escape_string($koneksi, $_POST['alamat']);
    $hp = mysqli_real_escape_string($koneksi, $_POST['no_hp']);
    $username = mysqli_real_escape_string($koneksi, $_POST['username']);
    $password = mysqli_real_escape_string($koneksi, $_POST['password']);
    $tgl = date('Y-m-d');

    // Cek username sudah dipakai atau belum
    $cek_user = mysqli_query($koneksi, "SELECT * FROM anggota WHERE username='$username'");
    if (mysqli_num_rows($cek_user) > 0) {
        $error = "Username sudah digunakan! Silakan pilih yang lain.";
    } else {
        // Query tanpa nomor_anggota (dibiarkan NULL)
        $query_insert = mysqli_query($koneksi, "INSERT INTO anggota (nama, jenis_kelamin, alamat, no_hp, tanggal_daftar, username, password, status) VALUES ('$nama', '$jk', '$alamat', '$hp', '$tgl', '$username', '$password', 'Pending')");
        if ($query_insert) {
            $sukses = "Pendaftaran berhasil! Silakan tunggu ACC dan pemberian Nomor Anggota dari Pustakawan.";
        } else {
            $error = "Pendaftaran gagal: " . mysqli_error($koneksi);
        }
    }
}

// PROSES LOGIN
if (isset($_POST['login'])) {
    $username = mysqli_real_escape_string($koneksi, $_POST['username']);
    $password = mysqli_real_escape_string($koneksi, $_POST['password']);
    $role_choice = $_POST['role_choice'];

    if ($role_choice == 'admin') {
        $result = mysqli_query($koneksi, "SELECT * FROM admin WHERE username='$username' AND password='$password' AND role='admin'");
        if (mysqli_num_rows($result) === 1) {
            $_SESSION['login'] = true;
            $_SESSION['username'] = $username;
            $_SESSION['role'] = 'admin';
            header("Location: dashboard.php");
            exit;
        } else {
            $error = "Login Admin Gagal! Periksa username/password.";
        }
    } else {
        // Cek ke tabel anggota
        $result = mysqli_query($koneksi, "SELECT * FROM anggota WHERE username='$username' AND password='$password'");
        if (mysqli_num_rows($result) === 1) {
            $row = mysqli_fetch_assoc($result);
            // Cek Status
            if ($row['status'] == 'Pending') {
                $error = "Akun Anda belum di-ACC oleh Pustakawan. Silakan hubungi petugas.";
            } else {
                $_SESSION['login'] = true;
                $_SESSION['username'] = $username;
                $_SESSION['role'] = 'user';
                header("Location: dashboard_user.php");
                exit;
            }
        } else {
            $error = "Login Pengguna Gagal! Cek username/password.";
        }
    }
}
