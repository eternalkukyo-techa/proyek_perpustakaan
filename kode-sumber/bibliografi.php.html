<?php
session_start();
if (!isset($_SESSION['login']) || $_SESSION['role'] != 'admin') { header("Location: index.php"); exit; }
include 'koneksi.php';

// Buat folder otomatis jika belum ada
if (!is_dir('covers')) { mkdir('covers', 0777, true); }
if (!is_dir('files')) { mkdir('files', 0777, true); }

// ================= PROSES TAMBAH BUKU =================
if (isset($_POST['tambah'])) {
    $kode = mysqli_real_escape_string($koneksi, $_POST['kode_buku']);
    $judul = mysqli_real_escape_string($koneksi, $_POST['judul']);
    $pengarang = mysqli_real_escape_string($koneksi, $_POST['pengarang']);
    $penerbit = mysqli_real_escape_string($koneksi, $_POST['penerbit']);
    $stok = (int)$_POST['stok'];
    
    // Upload Cover
    $nama_cover = null;
    if (isset($_FILES['cover_buku']) && $_FILES['cover_buku']['error'] === UPLOAD_ERR_OK) {
        $tmp_file = $_FILES['cover_buku']['tmp_name'];
        $ekstensi = strtolower(pathinfo($_FILES['cover_buku']['name'], PATHINFO_EXTENSION));
        if (in_array($ekstensi, ['jpg', 'jpeg', 'png'])) {
            $nama_cover = time() . '_cover.' . $ekstensi;
            move_uploaded_file($tmp_file, 'covers/' . $nama_cover);
        }
    }

    // Upload PDF
    $nama_pdf = null;
    if (isset($_FILES['file_pdf']) && $_FILES['file_pdf']['error'] === UPLOAD_ERR_OK) {
        $tmp_pdf = $_FILES['file_pdf']['tmp_name'];
        $ekstensi_pdf = strtolower(pathinfo($_FILES['file_pdf']['name'], PATHINFO_EXTENSION));
        if ($ekstensi_pdf == 'pdf') {
            $nama_pdf = time() . '_ebook.pdf';
            move_uploaded_file($tmp_pdf, 'files/' . $nama_pdf);
        }
    }

    $query = "INSERT INTO bibliografi (kode_buku, judul, pengarang, penerbit, stok, cover_buku, file_pdf) 
              VALUES ('$kode', '$judul', '$pengarang', '$penerbit', $stok, " . ($nama_cover ? "'$nama_cover'" : "NULL") . ", " . ($nama_pdf ? "'$nama_pdf'" : "NULL") . ")";
    mysqli_query($koneksi, $query);
    header("Location: bibliografi.php"); exit;
}

// ================= PROSES UPDATE/EDIT BUKU =================
if (isset($_POST['update'])) {
    $id = (int)$_POST['id_buku'];
    $kode = mysqli_real_escape_string($koneksi, $_POST['kode_buku']);
    $judul = mysqli_real_escape_string($koneksi, $_POST['judul']);
    $pengarang = mysqli_real_escape_string($koneksi, $_POST['pengarang']);
    $penerbit = mysqli_real_escape_string($koneksi, $_POST['penerbit']);
    $stok = (int)$_POST['stok'];

    // Ambil data lama untuk cek file
    $old_data = mysqli_fetch_assoc(mysqli_query($koneksi, "SELECT cover_buku, file_pdf FROM bibliografi WHERE id_buku=$id"));
    $nama_cover = $old_data['cover_buku'];
    $nama_pdf = $old_data['file_pdf'];

    // Cek jika ada upload cover baru
    if (isset($_FILES['cover_buku']) && $_FILES['cover_buku']['error'] === UPLOAD_ERR_OK) {
        $tmp_file = $_FILES['cover_buku']['tmp_name'];
        $ekstensi = strtolower(pathinfo($_FILES['cover_buku']['name'], PATHINFO_EXTENSION));
        if (in_array($ekstensi, ['jpg', 'jpeg', 'png'])) {
            if ($nama_cover && file_exists('covers/' . $nama_cover)) { unlink('covers/' . $nama_cover); } // Hapus cover lama
            $nama_cover = time() . '_cover.' . $ekstensi;
            move_uploaded_file($tmp_file, 'covers/' . $nama_cover);
        }
    }

    // Cek jika ada upload PDF baru
    if (isset($_FILES['file_pdf']) && $_FILES['file_pdf']['error'] === UPLOAD_ERR_OK) {
        $tmp_pdf = $_FILES['file_pdf']['tmp_name'];
        $ekstensi_pdf = strtolower(pathinfo($_FILES['file_pdf']['name'], PATHINFO_EXTENSION));
        if ($ekstensi_pdf == 'pdf') {
            if ($nama_pdf && file_exists('files/' . $nama_pdf)) { unlink('files/' . $nama_pdf); } // Hapus PDF lama
            $nama_pdf = time() . '_ebook.pdf';
            move_uploaded_file($tmp_pdf, 'files/' . $nama_pdf);
        }
    }

    $cover_sql = $nama_cover ? "'$nama_cover'" : "NULL";
    $pdf_sql = $nama_pdf ? "'$nama_pdf'" : "NULL";

    $query = "UPDATE bibliografi SET kode_buku='$kode', judul='$judul', pengarang='$pengarang', penerbit='$penerbit', stok=$stok, cover_buku=$cover_sql, file_pdf=$pdf_sql WHERE id_buku=$id";
    mysqli_query($koneksi, $query);
    header("Location: bibliografi.php"); exit;
}

}
