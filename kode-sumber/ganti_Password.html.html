    header("Location: index.php");

$tipe  = 'error';

    $username    = $_SESSION['username'];
    $pass_lama   = mysqli_real_escape_string($koneksi, $_POST['pass_lama']);
    $pass_baru   = mysqli_real_escape_string($koneksi, $_POST['pass_baru']);
    $pass_ulang  = mysqli_real_escape_string($koneksi, $_POST['pass_ulang']);

    $cek = mysqli_query($koneksi, "SELECT * FROM admin WHERE username='$username' AND password='$pass_lama'");

    if (mysqli_num_rows($cek) !== 1) {
        $pesan = "Password lama tidak sesuai!";
    } elseif ($pass_baru !== $pass_ulang) {
        $pesan = "Password baru dan ulangi password tidak sama!";
    } elseif (strlen($pass_baru) < 3) {
        $pesan = "Password baru minimal 3 karakter.";
    } else {
        mysqli_query($koneksi, "UPDATE admin SET password='$pass_baru' WHERE username='$username'");
        $pesan = "Password berhasil diubah! Silakan login ulang untuk memastikan.";
        $tipe  = 'sukses';
